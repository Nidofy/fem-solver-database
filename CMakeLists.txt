cmake_minimum_required(VERSION 3.24)
project(SolverDatabaseV2 VERSION 2.1.0 LANGUAGES CXX)
option(SDB_OPENMP "Enable OpenMP CPU execution" ON)
option(SDB_CUDA "Build native single GPU CUDA backend (CUDA 12+)" OFF)
option(SDB_SANITIZE "Enable address/undefined sanitizers on supported compilers" OFF)
add_library(solver_database_v2 INTERFACE)
target_include_directories(solver_database_v2 INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(solver_database_v2 INTERFACE cxx_std_20)
find_package(Threads REQUIRED)
target_link_libraries(solver_database_v2 INTERFACE Threads::Threads)
if(SDB_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(solver_database_v2 INTERFACE OpenMP::OpenMP_CXX)
endif()
if(SDB_SANITIZE)
  if(MSVC)
    target_compile_options(solver_database_v2 INTERFACE /fsanitize=address /Zi)
    target_link_options(solver_database_v2 INTERFACE /INCREMENTAL:NO)
  else()
    target_compile_options(solver_database_v2 INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(solver_database_v2 INTERFACE -fsanitize=address,undefined)
  endif()
endif()
if(SDB_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit 12 REQUIRED)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 70 80 90)
  endif()
  add_library(sdb_cuda STATIC src/cuda_backend.cu)
  target_link_libraries(sdb_cuda PUBLIC solver_database_v2 CUDA::cudart)
  target_compile_definitions(sdb_cuda PUBLIC SDB_HAS_CUDA=1)
  set_target_properties(sdb_cuda PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
  add_executable(sdb_cuda_tests tests/cuda_tests.cu)
  target_link_libraries(sdb_cuda_tests PRIVATE sdb_cuda)
  set_target_properties(sdb_cuda_tests PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
endif()
include(CTest)
add_library(sdb_lifecycle INTERFACE)
target_link_libraries(sdb_lifecycle INTERFACE solver_database_v2)
add_library(sdb_general_contact INTERFACE)
target_link_libraries(sdb_general_contact INTERFACE solver_database_v2)
add_executable(sdb_general_contact_tests tests/general_contact_tests.cpp)
target_link_libraries(sdb_general_contact_tests PRIVATE sdb_general_contact)
add_test(NAME general_contact_contracts COMMAND sdb_general_contact_tests ${CMAKE_CURRENT_BINARY_DIR}/general-contact-scratch)
add_executable(sdb_general_contact_demo examples/general_contact.cpp)
target_link_libraries(sdb_general_contact_demo PRIVATE sdb_general_contact)
add_test(NAME general_contact_demo COMMAND sdb_general_contact_demo ${CMAKE_CURRENT_BINARY_DIR}/general-contact-demo)
add_executable(sdb_contact_geometry_tests tests/general_contact_geometry_tests.cpp)
target_link_libraries(sdb_contact_geometry_tests PRIVATE sdb_general_contact)
add_test(NAME general_contact_geometry COMMAND sdb_contact_geometry_tests ${CMAKE_CURRENT_BINARY_DIR}/geometry-scratch)
add_executable(sdb_contact_domain_demo examples/general_contact_domain.cpp)
target_link_libraries(sdb_contact_domain_demo PRIVATE sdb_general_contact)
add_test(NAME general_contact_domain_demo COMMAND sdb_contact_domain_demo ${CMAKE_CURRENT_BINARY_DIR}/general-contact-domain-demo)
add_executable(sdb_run examples/sdb_run.cpp)
target_link_libraries(sdb_run PRIVATE sdb_lifecycle)
add_executable(sdb_lifecycle_tests tests/lifecycle_tests.cpp)
target_link_libraries(sdb_lifecycle_tests PRIVATE sdb_lifecycle)
add_test(NAME lifecycle_contracts COMMAND sdb_lifecycle_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/abaqus ${CMAKE_CURRENT_BINARY_DIR}/lifecycle-scratch)
add_executable(sdb_hex8_tests tests/hex8_tests.cpp)
target_link_libraries(sdb_hex8_tests PRIVATE sdb_lifecycle)
add_test(NAME hex8_contracts COMMAND sdb_hex8_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/abaqus ${CMAKE_CURRENT_BINARY_DIR}/hex8-scratch)
set_tests_properties(hex8_contracts PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=4" TIMEOUT 120)
add_executable(sdb_indexed_results_tests tests/indexed_results_tests.cpp)
target_link_libraries(sdb_indexed_results_tests PRIVATE sdb_lifecycle)
add_test(NAME indexed_results_contracts COMMAND sdb_indexed_results_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/abaqus/forced_tet.inp ${CMAKE_CURRENT_BINARY_DIR}/indexed-scratch)
set_tests_properties(indexed_results_contracts PROPERTIES TIMEOUT 300)
add_executable(sdb_async_results_tests tests/async_results_tests.cpp)
target_link_libraries(sdb_async_results_tests PRIVATE sdb_lifecycle)
add_test(NAME async_results_contracts COMMAND sdb_async_results_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/abaqus/forced_tet.inp ${CMAKE_CURRENT_BINARY_DIR}/async-scratch)
set_tests_properties(async_results_contracts PROPERTIES TIMEOUT 30)
add_executable(sdb_lifecycle_benchmark benchmarks/lifecycle_benchmark.cpp)
target_link_libraries(sdb_lifecycle_benchmark PRIVATE sdb_lifecycle)
foreach(target IN ITEMS sdb_hex8_tests sdb_indexed_results_tests sdb_async_results_tests sdb_run sdb_lifecycle_tests sdb_lifecycle_benchmark sdb_general_contact_tests sdb_general_contact_demo sdb_contact_geometry_tests sdb_contact_domain_demo)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME lifecycle_process_pipeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipeline.py $<TARGET_FILE:sdb_run> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/abaqus/forced_tet.inp ${CMAKE_CURRENT_BINARY_DIR}/pipeline-scratch)
endif()
foreach(target IN ITEMS sdb_tests explicit_tet sdb_benchmark)
  if(target STREQUAL "sdb_tests")
    set(source tests/tests.cpp)
  elseif(target STREQUAL "explicit_tet")
    set(source examples/explicit_tet.cpp)
  else()
    set(source benchmarks/benchmark.cpp)
  endif()
  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE solver_database_v2)
  if(SDB_CUDA)
    target_link_libraries(${target} PRIVATE sdb_cuda)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_test(NAME database_contracts COMMAND sdb_tests)
add_test(NAME explicit_numerics COMMAND explicit_tet)
if(SDB_CUDA)
  add_test(NAME cuda_contacts_memory COMMAND sdb_cuda_tests)
endif()
set_tests_properties(database_contracts explicit_numerics PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=4")
install(DIRECTORY include/ DESTINATION include)
install(TARGETS solver_database_v2 EXPORT SolverDatabaseV2Targets)
install(TARGETS sdb_lifecycle EXPORT SolverDatabaseV2Targets)
install(TARGETS sdb_general_contact EXPORT SolverDatabaseV2Targets)
install(TARGETS sdb_run RUNTIME DESTINATION bin)
if(SDB_CUDA)
  install(TARGETS sdb_cuda EXPORT SolverDatabaseV2Targets ARCHIVE DESTINATION lib)
endif()
install(EXPORT SolverDatabaseV2Targets DESTINATION lib/cmake/SolverDatabaseV2 NAMESPACE sdb::)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/SolverDatabaseV2Config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/SolverDatabaseV2Config.cmake INSTALL_DESTINATION lib/cmake/SolverDatabaseV2)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SolverDatabaseV2ConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SolverDatabaseV2Config.cmake ${CMAKE_CURRENT_BINARY_DIR}/SolverDatabaseV2ConfigVersion.cmake DESTINATION lib/cmake/SolverDatabaseV2)
