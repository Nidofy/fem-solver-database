cmake_minimum_required(VERSION 3.24)
project(InstalledDBF8 LANGUAGES CXX)
find_package(SolverDatabaseV2 2.4 CONFIG REQUIRED)
add_executable(installed_dbf8 main.cpp)
target_link_libraries(installed_dbf8 PRIVATE sdb::solver_database_v2 sdb::process_memory)
if(EXPECT_INDEX_BITS)
  target_compile_definitions(installed_dbf8 PRIVATE EXPECT_INDEX_BITS=${EXPECT_INDEX_BITS})
endif()
enable_testing()
add_test(NAME installed_dbf8 COMMAND installed_dbf8)
