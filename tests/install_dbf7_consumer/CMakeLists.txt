cmake_minimum_required(VERSION 3.24)
project(InstalledDBF7 LANGUAGES C CXX)
find_package(SolverDatabaseV2 2.3 CONFIG REQUIRED)
add_executable(installed_cpp main.cpp)
target_link_libraries(installed_cpp PRIVATE sdb::solver_database_v2)
add_executable(installed_c main.c)
target_link_libraries(installed_c PRIVATE sdb::sdb_reader)
enable_testing()
add_test(NAME installed_cpp_contracts COMMAND installed_cpp)
if(SDB_RESULT_FIXTURE)
  add_test(NAME installed_c_reader COMMAND installed_c "${SDB_RESULT_FIXTURE}")
endif()
